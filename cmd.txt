    -Pattern 'phpVersionList|checkPhpConf|phpVersionDir' |
    Select-Object Path, LineNumber, Line

Get-Process httpd | ForEach-Object {
    Write-Host "PID:" $_.Id

    try {
        $modules = @($_.Modules)
        Write-Host "Module Count:" $modules.Count

        $modules |
            Select-Object -First 10 ModuleName, FileName
    }
    catch {
        Write-Host "ERROR:" $_.Exception.Message
    }
}
