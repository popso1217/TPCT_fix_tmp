Get-ChildItem C:\wamp64 -Recurse -Filter *.php -File |
    Select-String -Pattern 'function\s+checkPhpConf' -Context 0,45



Select-String -Path "C:\wamp64\scripts\*.php" `
    -Pattern 'phpVersionList|checkPhpConf|phpVersionDir' |
    Select-Object Path, LineNumber, Line


C:\wamp64\bin\php\php8.5.12\php.exe -i 2>&1 |
    Select-String 'Warning|Error|Unable'



Get-CimInstance Win32_Process -Filter "name='httpd.exe'" |
    Select-Object ProcessId, CommandLine


Get-Process httpd | ForEach-Object {
    $p = $_

    try {
        $p.Modules |
            Where-Object {
                $_.ModuleName -match 'curl|ssl|crypto|ssh|nghttp|php'
            } |
            Select-Object `
                @{N='PID';E={$p.Id}},
                ModuleName,
                FileName,
                @{N='Version';E={$_.FileVersionInfo.FileVersion}}
    }
    catch {
        Write-Warning "無法讀取 PID $($p.Id) 的模組"
    }
} | Format-Table -AutoSize
