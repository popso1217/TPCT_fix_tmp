$apache = "C:\wamp64\bin\apache\apache2.4.65\bin"
$php = "C:\wamp64\bin\php\php8.5.12"

$dlls = @(
    "libssh2.dll",
    "libssl-3-x64.dll",
    "libcrypto-3-x64.dll",
    "nghttp2.dll",
    "libcurl.dll"
)

foreach ($dll in $dlls) {
    foreach ($dir in @($apache, $php)) {
        $file = Join-Path $dir $dll

        if (Test-Path $file) {
            Get-Item $file | Select-Object `
                Name, DirectoryName, Length,
                @{N="Version";E={$_.VersionInfo.FileVersion}}
        }
    }
}
